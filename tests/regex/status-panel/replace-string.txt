```html
<div id="zscd-test-status-panel-root"></div>
<script>
(function () {
  const root = document.getElementById('zscd-test-status-panel-root');
  if (!root || root.dataset.loading === 'true' || root.dataset.loaded === 'true') return;

  root.dataset.loading = 'true';

  fetch('https://raw.githubusercontent.com/yin-shi-luo142cm/zhongsheng-changdi-assets/4ce61772df018ee107a696906ea63ee17c9e6ceb/tests/regex/status-panel/index.html')
    .then(response => {
      if (!response.ok) throw new Error('HTTP ' + response.status);
      return response.text();
    })
    .then(markup => {
      const remote = new DOMParser().parseFromString(markup, 'text/html');
      if (!remote.querySelector('.status-panel')) {
        throw new Error('远端 HTML 缺少测试根节点 .status-panel');
      }

      remote.querySelectorAll('style').forEach((sourceStyle, index) => {
        const styleId = 'zscd-test-status-panel-style-' + index;
        if (document.getElementById(styleId)) return;

        const style = document.createElement('style');
        style.id = styleId;
        style.textContent = sourceStyle.textContent;
        document.head.appendChild(style);
      });

      root.innerHTML = remote.body.innerHTML;

      const sourceModule = remote.querySelector('script[type="module"]');
      if (sourceModule) {
        const module = document.createElement('script');
        module.type = 'module';
        module.textContent = sourceModule.textContent;
        document.body.appendChild(module);
      }

      root.dataset.loaded = 'true';
      root.dataset.loading = 'false';
    })
    .catch(error => {
      root.dataset.loading = 'false';
      root.dataset.failed = 'true';
      root.textContent = '测试介面载入失败，请检查网络或浏览器控制台。';
      console.error('[众生长地 regex test] remote resource load failed:', error);
    });
})();
</script>
```
